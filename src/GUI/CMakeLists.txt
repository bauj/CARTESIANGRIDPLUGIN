  ${QT_INCLUDES}
  ${OpenCASCADE_INCLUDE_DIR}

  ${KERNEL_INCLUDE_DIRS}
  ${GUI_INCLUDE_DIRS}
  ${GEOM_INCLUDE_DIRS}

  ${Boost_INCLUDE_DIRS}
  ${OMNIORB_INCLUDE_DIR}

  ${OMNIORB_DEFINITIONS}
  ${OpenCASCADE_DEFINITIONS}

  VoxelPluginGUI.cxx

  VoxelPlugin_images.ts
