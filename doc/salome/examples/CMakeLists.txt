# The examples shown in the documentation run as tests and are installed next to it, as in the other SMESH plugins.

INCLUDE(examples.set)

SET(COMPONENT_NAME CARTESIANGRIDPLUGIN)
SET(SALOME_TEST_DRIVER "$ENV{KERNEL_ROOT_DIR}/bin/salome/appliskel/salome_test_driver.py")
SET(TIMEOUT 300)

SET(TEST_INSTALL_DIRECTORY ${SALOME_CARTESIANGRIDPLUGIN_INSTALL_TESTS})

SALOME_GENERATE_TESTS_ENVIRONMENT(tests_env)

FOREACH(tfile ${EXAMPLE_NAMES})
  IF(SALOME_BUILD_TESTS)
    SET(TEST_NAME ${COMPONENT_NAME}_${tfile})
    ADD_TEST(NAME ${TEST_NAME}
             COMMAND ${PYTHON_EXECUTABLE} -B ${SALOME_TEST_DRIVER} ${TIMEOUT} ${CMAKE_CURRENT_SOURCE_DIR}/${tfile}.py)
    SET_TESTS_PROPERTIES(${TEST_NAME} PROPERTIES ENVIRONMENT "${tests_env}" LABELS "${COMPONENT_NAME}")
  ENDIF()
  INSTALL(FILES ${tfile}.py DESTINATION ${SALOME_INSTALL_DOC}/examples/CARTESIANGRIDPLUGIN)
  INSTALL(FILES ${tfile}.py DESTINATION ${TEST_INSTALL_DIRECTORY})
ENDFOREACH()

INSTALL(FILES examples.set DESTINATION ${TEST_INSTALL_DIRECTORY})
